-- 近 30 天各币种 GMV 与未结算金额
SELECT o.currency,
COUNT(*) AS order_cnt,
SUM(o.amount) AS gmv,
SUM(CASE WHEN o.status = 'PENDING'
THEN o.amount ELSE 0 END) AS unsettled
FROM orders o
JOIN customers c ON c.id = o.user_id
WHERE o.created_at >= NOW() - INTERVAL 30 DAY
AND c.re
GROUP BY o.currency
ORDER BY gmv DESC;
