生成的 SQL 可直接切到编辑器继续手改，改动不会被构建器覆盖
复制
SELECT o.currency, SUM(o.amount) AS gmv, COUNT(o.id) AS order_cnt, c.region_code
  FROM orders o
  INNER JOIN customers c ON c.id = o.user_id
  INNER JOIN order_items oi ON oi.order_id = o.id
  LEFT JOIN coupons cp ON cp.id = o.coupon_id
 WHERE o.created_at >= NOW() - INTERVAL 30 DAY AND c.region_code IN ('CN-SH', 'CN-BJ')
 GROUP BY o.currency, c.region_code
 ORDER BY o.currency ASC, gmv DESC;
